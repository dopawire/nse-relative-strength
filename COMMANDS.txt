================================================================
 NSE Relative-Strength view — command reference
================================================================

--- DAILY (website) ---------------------------------------------

# 1. fetch today's prices, rebuild data (~few min)
bash run_daily.sh

# 2. start the website
bash run_server.sh

# 3. open http://localhost:8000 in your browser
# (or click "Update Prices" / "Refresh Stocks" in the website header)

--- FIRST-TIME SETUP -------------------------------------------

# create the virtual environment
python3 -m venv .venv

# activate it (do this every new terminal session)
source .venv/bin/activate

# install dependencies
pip install -r requirements.txt

# install the browser used by refresh_classification.py
playwright install chromium


--- DAILY RUN (prices -> view) ---------------------------------

# activate first (if not already)
source .venv/bin/activate

# fetch latest Yahoo prices, append to history, rebuild rs_view.html
python3 build_rs.py

# then open the result
xdg-open rs_view.html


--- OTHER BUILD OPTIONS ----------------------------------------

# force a full re-fetch of ~2y history for every stock
python3 build_rs.py --refresh

# rebuild the HTML only, from cached prices (no fetching)
python3 build_rs.py --html-only

# quick test on the first 40 stocks
python3 build_rs.py --limit 40


--- REFRESH CLASSIFICATION (new NSE listings) ------------------

# list newly-listed symbols only, no browser
python3 refresh_classification.py --dry-run

# fetch + append the 4-level classification for new stocks
python3 refresh_classification.py

# same, but show the browser window (if Akamai blocks headless)
python3 refresh_classification.py --headful


--- DEACTIVATE -------------------------------------------------

deactivate
